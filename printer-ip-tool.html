2 功能设计与关键代码

四大功能
  1. 服务检查与启动：查询 Spooler（Print Spooler）服务状态；若未运行则把启动类型设为自动并启动。
  2. 列出本地网络打印机：遍历 HKLM\SYSTEM\CurrentControlSet\Control\Print\Printers 下各打印机，读取 Port，识别端口名含 192. / 10. / 172. 等 IP 段的标准 TCP/IP 端口，归为“网络打印机”。
  3. 修改 IP 并保存（v1.1 重写）：在 Standard TCP/IP Port\Port List 中查找已指向新 IP 的端口；若不存在则新建端口（以新 IP 命名，复制原端口全部配置，写入 PName/HostName），再把打印机的 Printers\<名>\Port 指向该端口。保存后自动重启 Print Spooler 使配置生效，并刷新列表且保持选中该打印机。
  4. 安装未列出打印机（v1.1 改为 exe）：点击后打开文件选择器，默认首选 *.exe（厂商驱动安装程序），选中即用 Process.Start 启动安装；同时保留 .zip（v4 驱动包，走 P/Invoke AddPrinterPackage）与 .inf（提示用系统向导）。

② 列出网络打印机（读注册表）

Printers.cs
using Microsoft.Win32;
public class PrinterInfo{ public string Name; public string Port; public string Ip; public bool IsNet; }
public static class Printers{
  static readonly string printersKey =
    @"SYSTEM\CurrentControlSet\Control\Print\Printers";
  public static List GetNetworkPrinters(){
    var list=new List();
    using var root=Registry.LocalMachine.OpenSubKey(printersKey);
    if(root==null) return list;
    foreach(var sub in root.GetSubKeyNames()){
      using var pk=root.OpenSubKey(sub);
      var port=pk?.GetValue("Port")?.ToString() ?? "";
      string ip=ExtractIp(port);
      if(!string.IsNullOrEmpty(ip)){
        list.Add(new PrinterInfo{Name=sub,Port=port,Ip=ip,IsNet=true});
      }
    }
    return list;
  }
  static string ExtractIp(string port){
    // 端口可能是 "192.168.1.50" 或 "192.168.1.50:9100" 或 "WSD"
    var m=System.Text.RegularExpressions.Regex.Match(port,
      @"(\d{1,3}\.){3}\d{1,3}");
    return m.Success? m.Value : "";
  }
}

③ 修改并保存 IP

IpChanger.cs
// 路径一（推荐）：在 Port List 中把原端口的 PName 改成新 IP，再同步打印机的 Port 值
public static void ChangePrinterIp(string printerName, string oldIp, string newIp, out string err){
  err="";
  const string mon = @"SYSTEM\CurrentControlSet\Control\Print\Monitors\Standard TCP/IP Port\Port List";
  string oldPortKey=null,newPortKey=null;
  using(var pl=Registry.LocalMachine.OpenSubKey(mon)){
    foreach(var pn in pl.GetSubKeyNames()){
      using var pk=pl.OpenSubKey(pn);
      var cur=pk?.GetValue("PName")?.ToString();
      if(cur==oldIp) oldPortKey=pn;
      if(cur==newIp) newPortKey=pn;
    }
  }
  // 若新端口不存在则创建（PName=newIp, Protocol=Raw/9100）
  if(newPortKey==null)
    newPortKey=CreatePort(newIp);
  // 更新打印机 Port 指向新端口
  using var pk=Registry.LocalMachine.OpenSubKey(
    @"SYSTEM\CurrentControlSet\Control\Print\Printers\\"+printerName,true);
  pk.SetValue("Port", newPortKey);
  // 可选：删除旧端口
  DeletePort(oldPortKey);
  err=string.Empty;
}

3 界面布局与交互流程

主界面自上而下分四个功能区，操作流以“刷新 → 选中 → 改 IP → 保存”为主：

  1. 服务区：状态灯（运行/停止）+ 检查并启动 Print Spooler 按钮（自动设为自动启动 + 启动）。
  2. 打印机列表区：DataGridView 显示 打印机名 / 端口 / 当前 IP / 类型 / 默认（默认打印机行标 ★ 默认）；刷新列表 重新扫描；「显示全部（含USB/虚拟）」复选框可列出本机全部打印机（非网络打印机标记“不可改IP”）；安装未列出打印机… 选择 .exe 安装程序并启动。
  3. 设为默认打印机：选中列表任意一台后点 设为默认打印机，调用 WMI Win32_Printer.SetDefaultPrinter()，失败再兜底 winspool SetDefaultPrinter；成功后列表刷新，★ 标记转移到新默认打印机。
  4. IP 修改区：当前 IP 只读显示 + 新 IP 输入框（带 IPv4 校验）+ 保存修改 按钮。
  5. 日志区：只读多行文本，记录每一步操作结果。
交互约定：未选中打印机或 服务未运行时，禁用“保存”；保存前弹出确认对话框显示旧→新 IP；保存成功后自动刷新列表并高亮该行。

        static bool IsValidIp(string ip)
        {
            if (string.IsNullOrWhiteSpace(ip)) return false;
            var parts = ip.Split('.');
            if (parts.Length != 4) return false;
            foreach (var p in parts)
            {
                int n;
                if (!int.TryParse(p, out n) || n < 0 || n > 255) return false;
            }
            return true;
        }

    public class PrinterInfo
    {
        public string Name, Port, Ip;
        public bool IsNet = true;
    }

    public static class Printers
    {
        private static readonly string _k =
            @"SYSTEM\CurrentControlSet\Control\Print\Printers";
        public static List GetNetworkPrinters()
        {
            var list = new List();
            using (var root = Registry.LocalMachine.OpenSubKey(_k))
            {
                if (root == null) return list;
                foreach (var sub in root.GetSubKeyNames())
                {
                    string port;
                    using (var pk = root.OpenSubKey(sub))
                        port = pk?.GetValue("Port")?.ToString() ?? "";
                    string ip = ExtractIp(port);
                    if (!string.IsNullOrEmpty(ip))
                        list.Add(new PrinterInfo { Name = sub, Port = port, Ip = ip });
                }
            }
            return list;
        }
        public static string ExtractIp(string port)
        {
            var m = Regex.Match(port, @"(\d{1,3}\.){3}\d{1,3}");
            return m.Success ? m.Value : "";
        }
    }

    public static class IpChanger
    {
        private static readonly string _mon =
            @"SYSTEM\CurrentControlSet\Control\Print\Monitors\Standard TCP/IP Port\Port List";

        public static void ChangePrinterIp(string printer, string oldIp, string newIp, out string err)
        {
            err = "";
            string oldPortKey = null, newPortKey = null;
            using (var pl = Registry.LocalMachine.OpenSubKey(_mon))
            {
                foreach (var pn in pl.GetSubKeyNames())
                {
                    using (var pk = pl.OpenSubKey(pn))
                    {
                        var cur = pk?.GetValue("PName")?.ToString();
                        if (cur == oldIp) oldPortKey = pn;
                        if (cur == newIp) newPortKey = pn;
                    }
                }
            }
            if (newPortKey == null)
                newPortKey = CreatePort(newIp);
            using (var pk = Registry.LocalMachine.OpenSubKey(
                @"SYSTEM\CurrentControlSet\Control\Print\Printers\\" + printer, true))
            {
                if (pk == null) { err = "未找到打印机注册表项：" + printer; return; }
                pk.SetValue("Port", newPortKey);
            }
            if (oldPortKey != null && oldPortKey != newPortKey)
                DeletePort(oldPortKey);
        }

        private static string CreatePort(string ip)
        {
            // 端口名规则：以 P 开头的唯一名，这里用时间戳避免冲突
            string name = "P" + DateTime.Now.ToString("yyyyMMddHHmmss");
            using (var pl = Registry.LocalMachine.CreateSubKey(_mon))
            using (var pk = pl.CreateSubKey(name))
            {
                pk.SetValue("PName", ip);
                pk.SetValue("Protocol", "Standard (9100)");
            }
            return name;
        }
        private static void DeletePort(string name)
        {
            Registry.LocalMachine.DeleteSubKeyTree(_mon + "\\" + name, false);
        }
    }

    public static class Installer
    {
        [DllImport("winspool.drv", CharSet = CharSet.Unicode, SetLastError = true)]

        public static void InstallDriver(string path, out string msg)
        {
            msg = "";
            try
            {
                if (path.EndsWith(".zip", StringComparison.OrdinalIgnoreCase))

7 常见问题与注意事项

现象处理
保存后打印机电源/驱动未自动跟随 IP本工具改的是端口 PName 与打印机绑定，不驱动打印机硬件；需在新 IP 上保证打印机仍在网段，或在打印机面板确认 IP 已更新。改完可“测试打印”。
只列出 1 台网络打印机，是不是漏了？通常没有漏。工具只列“端口含 IP 的 TCP/IP 网络打印机”，USB、PDF/虚拟打印机本来就没有 IP，会被正确过滤。诊断示例（本机 4 台打印机）：Brother HL-B2050DN（端口 192.168.10.17）✅ 可改 IP；HP ColorLaserJet（USB001）、Microsoft Print to PDF（PORTPROMPT:）、导出为WPS PDF（Kingsoft 虚拟端口）❌ 无 IP。勾选「显示全部」即可看到全部 4 台及其类型，日志也会提示“共 N 台，网络 M 台”。
列表里出现 WSD / USB 打印机本工具只识别“端口名含 IPv4 数字段”或 Port List 中 PName/HostName 含 IP 的项；WSD、USB、Local Port 等非网络端口自动过滤（可用「显示全部」查看）。
日志出现 “创建端口失败（status=5，拒绝访问）”并非没以管理员运行。XcvDataW 的 AddPort 要求 SeLoadDriverPrivilege 处于“已启用”状态（管理员令牌里默认存在但被禁用），且句柄需带 SERVER_ACCESS_ADMINISTER。v1.2 已自动启用该特权并以管理员权限句柄调用；若仍失败会自动回退到 PowerShell Add-PrinterPort / Set-Printer（系统自带，继承已提升令牌），再兜底注册表。启动日志会显示当前是否真的处于已提升会话。
报 “无法访问/拒绝访问”未以管理员运行。请右键“以管理员身份运行”（清单要求管理员时系统会弹 UAC）。
驱动包安装失败确认 .zip 是厂商 v4 驱动包（解压后有 driver.inf / x64 目录），Win7 需 .NET 4.0；老系统对 v4 包支持不佳，可改用 .inf 向导安装。
服务设为自动后开机仍未自启个别精简版系统缺 Spooler 依赖；确认 services.msc 中 Print Spooler 已设为自动并启动。
合规与风险：修改打印端口注册表属于系统级写操作，建议在受控环境中做；生产网络批量变更前先备份 HKLM\SYSTEM\CurrentControlSet\Control\Print 相关键（可用 regedit 导出 .reg 留底）。安装驱动包前请确认来源可信，避免植入恶意驱动。
